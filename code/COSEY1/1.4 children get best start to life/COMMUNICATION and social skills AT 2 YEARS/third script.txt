and cast(numericvalue as float64) >35 then 'Child devlopement on schedule'

 ,

 -- CAST(age_at_encounter AS INT64) AS age_at_encounter,

 case when  ctv3text =  'ASQ-3 24 month questionnaire - personal-social sco' then 

	
'ASQ-3 24 month questionnaire - personal-social score'

end as ASQ,

  numericvalue,

  cast(event_year as string) event_year

  `yhcr-prd-phm-bia-core.CB_1935_AK.src_bmbc_HVDatap2`
