create view `yhcr-prd-phm-bia-core.CB_1935_AK.src_bmbc_HV68imd`

as
  SELECT
    a.*,
   
    c.lsoa,
    d.partialpostcode,
    d.tbl_srpatientaddresshistory_start_date,
    d.tbl_srpatientaddresshistory_end_date,
    e.Index_of_Multiple_Deprivation_Decile as IMD,
    e.LSOA_Name,
   -- EXTRACT(YEAR FROM a.dateevent) as event_year,
    ROW_NUMBER() OVER (PARTITION BY a.person_id, a.dateevent ORDER BY a.person_id, a.dateevent) AS row_num
  FROM
`yhcr-prd-phm-bia-core.CB_1935_AK.src_bmbc_HV68Datap2` a
  INNER JOIN
    `yhcr-prd-phm-bia-core.CB_LOOKUPS.tbl_person_lsoa` c ON a.person_id = c.person_id
  INNER JOIN
    `yhcr-prd-phm-bia-core.CB_FDM_PrimaryCare_V8.tbl_srpatientaddresshistory` d ON a.person_id = d.person_id
  INNER JOIN
    `yhcr-prd-phm-bia-core.CB_LOOKUPS.tbl_IMD_by_LSOA` e ON c.lsoa = e.LSOA_code
 
   AND e.LSOA_Name LIKE 'Bradford%'
    AND EXTRACT(YEAR FROM a.dateevent) >= 2012
